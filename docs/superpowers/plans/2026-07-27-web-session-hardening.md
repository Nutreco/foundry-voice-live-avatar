# Web Session Hardening Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix the browser audio contract at 24 kHz, make browser session cleanup and reconnection deterministic, throttle login attempts, fail fast on invalid configuration, remove unsupported turn-taking settings, and cover browser lifecycle behavior with Playwright.

**Architecture:** Keep the existing ASP.NET bridge and thin TypeScript client. Centralize the audio rate in a server constant, validate JSON before building `ConfigState`, add a named login rate-limit policy, and make the browser client own one idempotent disconnect path that releases all resources before a fresh reconnect. Playwright runs the built static UI with mocked browser media, WebRTC, and WebSocket APIs.

**Tech Stack:** .NET 10 minimal APIs, ASP.NET Core rate limiting and health checks, Azure Voice Live SDK, TypeScript 7, esbuild, Playwright, xUnit.

In `ThinVoiceLiveClient`, add:

```typescript
private disconnecting = false;

private async disconnect(message?: string) {
  if (this.disconnecting) return;
  this.disconnecting = true;

  if (this.pingId) {
    window.clearInterval(this.pingId);
    this.pingId = 0;
  }

  this.streamingMic = false;
  this.micStream?.getTracks().forEach((track) => track.stop());
  this.micStream = undefined;

  for (const node of this.audioNodes) node.disconnect();
  this.audioNodes = [];

  const audioContext = this.audioContext;
  this.audioContext = undefined;
  if (audioContext && audioContext.state !== "closed")
    await audioContext.close();

  this.pc?.close();
  this.pc = undefined;
  this.view.avatar.srcObject = null;

  const socket = this.socket;
  this.socket = undefined;
  if (socket && socket.readyState < WebSocket.CLOSING)
    socket.close();

  this.readyConfig = undefined;
  this.interactive?.setReady(false);
  this.interactive?.setDisconnected(true);
  if (message) this.fail(message);
  this.setStatus("connection", "disconnected");
}
```

Cleanup must tolerate partially initialized resources and repeated calls.

import type { Page } from "@playwright/test";

type MockOptions = {
  rejectMicrophone?: boolean;
};

export type LifecycleState = {
  socketCount: number;
  tracksStopped: number;
  audioContextsClosed: number;
  peerConnectionsClosed: number;
  requestedAudioRates: number[];
};

export async function installBrowserMocks(
  page: Page,
  options: MockOptions = {},
) {
  await page.addInitScript(({ rejectMicrophone }) => {
    const state = {
      sockets: [] as MockWebSocket[],
      tracksStopped: 0,
      audioContextsClosed: 0,
      peerConnectionsClosed: 0,
      requestedAudioRates: [] as number[],
    };

    class MockWebSocket extends EventTarget {
      static CONNECTING = 0;
      static OPEN = 1;
      static CLOSING = 2;
      static CLOSED = 3;
      readyState = MockWebSocket.OPEN;
      binaryType = "";
      closeCalls = 0;
      sent: unknown[] = [];

      constructor(_url: string) {
        super();
        state.sockets.push(this);
        queueMicrotask(() => this.dispatchEvent(new Event("open")));
      }

      send(value: unknown) {
        this.sent.push(value);
      }

      close() {
        this.closeCalls++;
        this.readyState = MockWebSocket.CLOSED;
      }
    }

    Object.assign(window, { WebSocket: MockWebSocket });

    const track = { stop: () => state.tracksStopped++ };
    Object.defineProperty(navigator, "mediaDevices", {
      value: {
        getUserMedia: async () => {
          if (rejectMicrophone) throw new Error("microphone unavailable");
          return { getTracks: () => [track] };
        },
      },
    });

    class MockAudioNode {
      connect<T>(target: T): T {
        return target;
      }

      disconnect() {}
    }

    class MockAudioContext {
      sampleRate = 48_000;
      state: AudioContextState = "suspended";
      destination = new MockAudioNode();
      audioWorklet = { addModule: async (_url: string) => {} };

      constructor(init?: AudioContextOptions) {
        if (init?.sampleRate) state.requestedAudioRates.push(init.sampleRate);
      }

      createMediaStreamSource(_stream: MediaStream) {
        return new MockAudioNode();
      }

      createGain() {
        return Object.assign(new MockAudioNode(), { gain: { value: 1 } });
      }

      async resume() {
        this.state = "running";
      }

      async close() {
        if (this.state === "closed") return;
        this.state = "closed";
        state.audioContextsClosed++;
      }
    }

    class MockAudioWorkletNode extends MockAudioNode {
      port = {
        onmessage: null as ((event: MessageEvent<ArrayBuffer>) => void) | null,
      };

      constructor(_context: AudioContext, _name: string) {
        super();
      }
    }

    class MockPeerConnection extends EventTarget {
      iceGatheringState: RTCIceGatheringState = "complete";
      connectionState: RTCPeerConnectionState = "new";
      localDescription: RTCSessionDescriptionInit | null = null;
      ontrack: ((event: RTCTrackEvent) => void) | null = null;
      onconnectionstatechange: (() => void) | null = null;

      constructor(_configuration?: RTCConfiguration) {
        super();
      }

      addTransceiver(_kind: string, _init?: RTCRtpTransceiverInit) {}

      async createOffer(): Promise<RTCSessionDescriptionInit> {
        return { type: "offer", sdp: "mock-offer" };
      }

      async setLocalDescription(description: RTCSessionDescriptionInit) {
        this.localDescription = description;
      }

      async setRemoteDescription(_description: RTCSessionDescriptionInit) {}

      close() {
        if (this.connectionState === "closed") return;
        this.connectionState = "closed";
        state.peerConnectionsClosed++;
      }
    }

    Object.assign(window, {
      AudioContext: MockAudioContext,
      AudioWorkletNode: MockAudioWorkletNode,
      RTCPeerConnection: MockPeerConnection,
    });

    HTMLMediaElement.prototype.play = async () => {};

    Object.assign(window, {
      __voiceLiveTest: {
        send(frame: unknown) {
          const socket = state.sockets.at(-1);
          socket?.dispatchEvent(new MessageEvent("message", {
            data: JSON.stringify(frame),
          }));
        },
        close(wasClean: boolean) {
          const socket = state.sockets.at(-1);
          if (!socket) throw new Error("No WebSocket exists.");
          socket.readyState = MockWebSocket.CLOSED;
          socket.dispatchEvent(new CloseEvent("close", {
            code: wasClean ? 1000 : 1006,
            wasClean,
          }));
        },
        state(): LifecycleState {
          return {
            socketCount: state.sockets.length,
            tracksStopped: state.tracksStopped,
            audioContextsClosed: state.audioContextsClosed,
            peerConnectionsClosed: state.peerConnectionsClosed,
            requestedAudioRates: [...state.requestedAudioRates],
          };
        },
      },
    });
  }, { rejectMicrophone: options.rejectMicrophone ?? false });
}

export async function sendServerFrame(page: Page, frame: unknown) {
  await page.evaluate((value) => {
    (window as any).__voiceLiveTest.send(value);
  }, frame);
}

export async function sendReady(page: Page) {
  await sendServerFrame(page, {
    t: "ready",
    config: {
      mode: "model",
      activeMode: "gated",
      agentName: "test-agent",
      safeQuestions: [],
      avatarCharacter: "lisa",
      avatarStyle: "casual-sitting",
    },
    iceServers: [],
  });
}

export async function closeLatestSocket(page: Page, wasClean: boolean) {
  await page.evaluate((clean) => {
    (window as any).__voiceLiveTest.close(clean);
  }, wasClean);
}

export async function lifecycle(page: Page): Promise<LifecycleState> {
  return page.evaluate(() => (window as any).__voiceLiveTest.state());
}
